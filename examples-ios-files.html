Headless TreeController

This entire UI is rendered without <web-treeview>. The tree data engine (TreeController) handles the hierarchy, and all rendering is custom vanilla JS.

No web component used
TreeController is imported directly and used as a headless data engine. The iOS phone frame, navigation, search, sort, file list, and detail sheet are all custom DOM code.

Setup

import { TreeController } from './dist/web-treeview.js';

const ctrl = new TreeController({
  data: notepadppData,
  idMember: 'id',
  pathMember: 'path',
  treePathSeparator: '/',
  expandLevel: 99,
});

Getting Children

// Get direct children of current folder
let children = ctrl.getChildren(currentPath);

// Filter by search text
if (searchText) {
  const q = searchText.toLowerCase();
  children = children.filter(n =>
    (n.data?.name || '').toLowerCase().includes(q)
  );
}

Folder Navigation

let currentPath = 'Notepad++';
let navStack = []; // internal back-stack

function navigateTo(path) {
  navStack.push(currentPath);
  currentPath = path;
  render();
}

// Back button
backBtn.addEventListener('click', () => {
  if (navStack.length > 0) {
    currentPath = navStack.pop();
    render();
  }
});

Rendering a File Row

for (const node of children) {
  const row = document.createElement('button');
  row.className = 'ios-file-row';

  // Icon from extension
  const icon = document.createElement('span');
  icon.textContent = getFileIcon(node);

  // Name + metadata
  const name = document.createElement('span');
  name.textContent = node.data?.name;

  const meta = document.createElement('span');
  if (node.data?.isDirectory) {
    const items = ctrl.getChildren(node.path);
    meta.textContent = items.length + ' items';
  } else {
    meta.textContent = formatSize(node.data.size);
  }

  // Navigate into folder or show detail
  if (node.data?.isDirectory) {
    row.addEventListener('click',
      () => navigateTo(node.path));
  } else {
    row.addEventListener('click',
      () => showDetail(node));
  }
}

Sorting

function sortNodes(nodes) {
  return [...nodes].sort((a, b) => {
    // Folders always first
    const aDir = a.data?.isDirectory ? 0 : 1;
    const bDir = b.data?.isDirectory ? 0 : 1;
    if (aDir !== bDir) return aDir - bDir;

    // Then sort by selected field
    switch (sortBy) {
      case 'name':
        return a.data.name.localeCompare(b.data.name);
      case 'size':
        return a.data.size - b.data.size;
      // ...
    }
  });
}

APIs Used

  • ctrl.getChildren(path) — folder contents
  • ctrl.getNodeByPath(path) — node lookup
  • node.data — original user data
  • node.path — LTree path
  • node.hasChildren — folder check
